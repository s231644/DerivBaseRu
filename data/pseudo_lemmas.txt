# Псевдо-леммы: связанные морфемы, устаревшие/фольклорные варианты и битые заголовки.
# Исключаются из исходных слов пайплайна, чтобы от них не порождались производные
# (дцать -> дцатеро -> вдцатером и т.п.). Формат: одно слово в строке, # комментарий.
дцать        # связанная морфема (суффикс -дцать), не самостоятельное числительное
надцать      # связанная морфема (суффикс -надцать)
осемь        # диал./устар. вариант «восемь»
восьмнадцать # устар. вариант «восемнадцать»
осьмнадцать  # устар. вариант «восемнадцать»
дванадесять  # устар. «двенадцать»
тридевять    # фольклорное (тридевять земель)
двапять      # битый заголовок
…ять         # битый заголовок (многоточие вместо буквы)
